SELECT
    cs.ArcadiaCredentialId AS id,
    cs.ArcadiaCredentialId AS credentialId,
    cs.CorrelationId AS correlationId,
    cs.Status AS status,
    cs.Status AS credentialStatus,
    cs.StatusDetail AS statusDetail,
    cs.StatusDetail AS credentialStatusDetail,
    cs.ProviderId AS providerId,
    cs.IsActive AS isActive,
    cs.IsCustomerActionRequired AS isCustomerActionRequired,
    cs.NextScheduledAccountRunAt AS nextScheduledAccountRunAt,
    cs.NextScheduledAccountRunAt AS nextRunAt,
    cs.LastCredentialSyncedOn AS syncedOn,
    cs.LastCredentialSyncedOn AS createdAt,
    c.ClientName AS client_name
FROM arcadia.ArcadiaCredentialState cs
INNER JOIN dbo.Client c ON cs.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{ appsmith.store.credentialSearch?.clientId || "" }}'
  AND cs.IsCustomerActionRequired = (
    CASE WHEN '{{ appsmith.store.credentialSearch?.source || "" }}' = 'Arcadia' THEN 0 ELSE 1 END
  )
  AND cs.Status IN (
    SELECT value FROM STRING_SPLIT('{{ CredentialsUtils.getStatusList() }}', ',')
  )
ORDER BY cs.Status, cs.ArcadiaCredentialId;
