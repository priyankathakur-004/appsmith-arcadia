SELECT
    ac.ArcadiaCredentialId AS id,
    ac.ArcadiaCredentialId AS credentialId,
    ac.AccountNumber AS accountNumber,
    ac.Status AS status,
    ac.Status AS credentialStatus,
    ac.StatusDetail AS statusDetail,
    ac.StatusDetail AS credentialStatusDetail,
    ac.ProviderId AS providerId,
    ac.IsActive AS isActive,
    ac.IsCustomerActionRequired AS isCustomerActionRequired,
    ac.NextScheduledAccountRunAt AS nextScheduledAccountRunAt,
    ac.NextScheduledAccountRunAt AS nextRunAt,
    ac.CreatedAt AS createdAt,
    ac.SyncedOn AS syncedOn,
    c.ClientName AS client_name
FROM dbo.ArcadiaCredential ac
INNER JOIN dbo.Client c ON ac.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{ appsmith.store.credentialSearch?.clientId || "" }}'
  AND ac.IsCustomerActionRequired = (
    CASE WHEN '{{ appsmith.store.credentialSearch?.source || "" }}' = 'Arcadia' THEN 0 ELSE 1 END
  )
  AND ac.Status IN (
    SELECT value FROM STRING_SPLIT('{{ CredentialsUtils.getStatusList() }}', ',')
  )
ORDER BY ac.Status, ac.AccountNumber;
