SELECT
    aca.IdClient AS idClient,
    c.ClientName AS client_name,
    aca.IdClientAccount AS idClientAccount,
    aca.AccountNumber AS accountNumber,
    aca.NormalizedAccountNumber AS normalizedAccountNumber,
    aca.ArcadiaAccountId AS id,
    aca.ArcadiaAccountId AS accountId,
    aca.ArcadiaCredentialId AS credentialId,
    aca.ProviderId AS providerId,
    aca.ProviderName AS providerName,
    aca.LatestStatementDate AS latestStatementDate,
    aca.NextExpectedPostDate AS nextExpectedPostDate,
    aca.Status AS status,
    aca.Status AS accountStatus,
    aca.StatusDetail AS statusDetail,
    aca.StatusDetail AS accountStatusDetail,
    aca.AccountType AS type,
    aca.AccountType AS accountType,
    aca.IsActive AS isActive,
    aca.MissingSince AS missingSince,
    aca.LastSeenOn AS lastSeenOn,
    aca.LastSyncedOn AS syncedOn
FROM arcadia.ArcadiaCredentialAccount aca
INNER JOIN dbo.Client c ON aca.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{ appsmith.store.accountsSearch?.clientId || "" }}'
  AND aca.Status IN (
    SELECT value FROM STRING_SPLIT('{{ AccountsUtils.getStatusList() }}', ',')
  )
ORDER BY aca.Status, aca.AccountNumber;
