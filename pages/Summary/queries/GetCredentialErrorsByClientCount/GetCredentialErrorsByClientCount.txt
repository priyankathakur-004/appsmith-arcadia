SELECT COUNT(*) AS total
FROM arcadia.ArcadiaCredentialState cs
INNER JOIN dbo.Client c ON cs.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{ appsmith.store.credentialSearch?.clientId || "" }}'
  AND cs.IsCustomerActionRequired = (
    CASE WHEN '{{ appsmith.store.credentialSearch?.source || "" }}' = 'Arcadia' THEN 0 ELSE 1 END
  )
  AND cs.Status IN (
    SELECT value FROM STRING_SPLIT('{{ CredentialsUtils.getStatusList() }}', ',')
  );
