SELECT COUNT(*) AS total
FROM arcadia.ArcadiaCredentialAccount aca
INNER JOIN dbo.Client c ON aca.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{ appsmith.store.accountsSearch?.clientId || "" }}'
  AND aca.Status IN (
    SELECT value FROM STRING_SPLIT('{{ AccountsUtils.getStatusList() }}', ',')
  );
