SELECT
    ap.IdVendor AS IdVendor,
    ap.VendorName AS VendorName,
    ap.ArcadiaProviderName AS ProviderName,
    ap.Country AS Country,
    ap.Region AS Region,
    ap.Website AS Website,
    ap.ArcadiaProviderId AS ProviderId,
    ap.SyncedOn AS SyncedOn
FROM dbo.ArcadiaProviders ap
WHERE 1=1
  AND (
    '{{ProvVendorInput.text || ""}}' = ''
    OR LOWER(LTRIM(RTRIM(ISNULL(ap.VendorName, '')))) LIKE '%' + LOWER('{{ProvVendorInput.text || ""}}') + '%'
    OR LOWER(LTRIM(RTRIM(ISNULL(ap.ArcadiaProviderName, '')))) LIKE '%' + LOWER('{{ProvVendorInput.text || ""}}') + '%'
  )
ORDER BY ap.VendorName
OFFSET {{(ProvidersTable.pageNo - 1) * 50}} ROWS
FETCH NEXT 50 ROWS ONLY;
