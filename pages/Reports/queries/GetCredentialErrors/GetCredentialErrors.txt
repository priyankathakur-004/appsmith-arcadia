SELECT
    ac.ArcadiaCredentialId AS arcadia_credential_id,
    ac.AccountNumber AS account_number,
    ac.Status AS status,
    ac.StatusDetail AS status_detail,
    ac.ProviderId AS provider_id,
    ac.IsActive AS is_active,
    ac.IsCustomerActionRequired AS is_customer_action_required,
    ac.NextScheduledAccountRunAt AS next_scheduled_account_run_at,
    ac.CreatedAt AS created_at,
    ac.SyncedOn AS synced_on,
    c.ClientName AS client_name
FROM dbo.ArcadiaCredential ac
INNER JOIN dbo.Client c ON ac.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{appsmith.store.credentialSearch?.clientId || ""}}'
  AND (
    ('{{appsmith.store.credentialSearch?.source || ""}}' = 'US'
      AND ac.IsCustomerActionRequired = 1
      AND ac.Status IN ('Connection Failure', 'Connection In Progress'))
    OR
    ('{{appsmith.store.credentialSearch?.source || ""}}' = 'Arcadia'
      AND ac.IsCustomerActionRequired = 0
      AND ac.Status IN ('Connection Deactivated', 'Connection In Progress', 'Connection Failure'))
  )
ORDER BY ac.Status, ac.AccountNumber
OFFSET {{CredErrorsTable.pageOffset}} ROWS
FETCH NEXT {{CredErrorsTable.pageSize}} ROWS ONLY;
