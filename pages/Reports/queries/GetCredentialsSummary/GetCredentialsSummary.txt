SELECT
    c.ClientName AS Customers,
    c.ProviderClientId AS ID,
    COUNT(ac.IdArcadiaCredential) AS ArcadiaFeed,
    SUM(CASE WHEN ac.Status = 'CONNECTION_SUCCESS' THEN 1 ELSE 0 END) AS Synced,
    CASE
        WHEN COUNT(ac.IdArcadiaCredential) > 0
        THEN CAST(ROUND(SUM(CASE WHEN ac.Status = 'CONNECTION_SUCCESS' THEN 1.0 ELSE 0 END) / COUNT(ac.IdArcadiaCredential) * 100, 2) AS VARCHAR) + '%'
        ELSE '0%'
    END AS Percentage,
    CASE
        WHEN COUNT(ac.IdArcadiaCredential) > 0
        THEN CAST(ROUND(100.0 - (SUM(CASE WHEN ac.Status = 'CONNECTION_SUCCESS' THEN 1.0 ELSE 0 END) / COUNT(ac.IdArcadiaCredential) * 100), 2) AS VARCHAR) + '%'
        ELSE '100%'
    END AS ToFix,
    SUM(CASE WHEN ac.IsCustomerActionRequired = 1 AND ac.Status IN ('CONNECTION_FAILURE', 'CONNECTION_IN_PROGRESS') THEN 1 ELSE 0 END) AS US,
    SUM(CASE WHEN ac.IsCustomerActionRequired = 0 AND ac.Status IN ('CONNECTION_DEACTIVATED', 'CONNECTION_IN_PROGRESS', 'CONNECTION_FAILURE') THEN 1 ELSE 0 END) AS Arcadia
FROM dbo.ArcadiaCredential ac
INNER JOIN dbo.Client c ON ac.IdClient = c.IdClient
GROUP BY c.ClientName, c.ProviderClientId, c.IdClient
ORDER BY c.ClientName;
