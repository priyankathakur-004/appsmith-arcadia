SELECT COUNT(*) AS total
FROM dbo.ArcadiaCredential ac
INNER JOIN dbo.Client c ON ac.IdClient = c.IdClient
WHERE c.ProviderClientId = '{{appsmith.store.credentialSearch?.clientId || ""}}'
  AND (
    ('{{appsmith.store.credentialSearch?.source || ""}}' = 'US'
      AND ac.IsCustomerActionRequired = 1
      AND ac.Status IN ('Connection Failure', 'Connection In Progress'))
    OR
    ('{{appsmith.store.credentialSearch?.source || ""}}' = 'Arcadia'
      AND ac.IsCustomerActionRequired = 0
      AND ac.Status IN ('Connection Deactivated', 'Connection In Progress', 'Connection Failure'))
  );
