SELECT
    COUNT(1) AS total,
    SUM(CASE WHEN cs.Status = 'CONNECTION_SUCCESS' THEN 1 ELSE 0 END) AS connected,
    SUM(CASE WHEN cs.Status = 'CONNECTION_IN_PROGRESS' THEN 1 ELSE 0 END) AS pending,
    SUM(CASE WHEN cs.Status IN ('CONNECTION_FAILURE', 'CONNECTION_DEACTIVATED') THEN 1 ELSE 0 END) AS failed,
    SUM(CASE WHEN cs.Status IS NULL THEN 1 ELSE 0 END) AS missing,
    SUM(CASE WHEN cs.IsCustomerActionRequired = 1 THEN 1 ELSE 0 END) AS customer_action_required
FROM arcadia.ArcadiaCredentialState cs
WHERE (
    '{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
);
