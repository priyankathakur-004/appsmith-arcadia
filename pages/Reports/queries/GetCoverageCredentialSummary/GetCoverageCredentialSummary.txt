SELECT
    SUM(CASE
        WHEN EXISTS (
            SELECT 1 FROM arcadia.ArcadiaCredentialAccount aca
            WHERE aca.IdClient = cs.IdClient
              AND aca.ArcadiaCredentialId = cs.ArcadiaCredentialId
              AND ISNULL(aca.IsActive, 0) = 1
        ) THEN 1 ELSE 0
    END) AS connected_in_arcadia,
    SUM(CASE WHEN cs.Status = 'CONNECTION_IN_PROGRESS' THEN 1 ELSE 0 END) AS pending,
    SUM(CASE
        WHEN cs.Status = 'CONNECTION_FAILURE'
         AND NOT (
            LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%mfa%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multi-factor%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multi_factor%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multifactor%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%two-factor%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%two_factor%'
            OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%otp%'
         )
        THEN 1 ELSE 0
    END) AS connection_failures,
    SUM(CASE
        WHEN LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%mfa%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multi-factor%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multi_factor%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%multifactor%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%two-factor%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%two_factor%'
          OR LOWER(ISNULL(cs.StatusDetail, '')) LIKE '%otp%'
        THEN 1 ELSE 0
    END) AS mfa_failures,
    COUNT(1) AS total
FROM arcadia.ArcadiaCredentialState cs
WHERE (
    '{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
);
