-- Credentials summary mirroring DSS_API _build_credential_status_by_id +
-- credentialOverview* computed values in SyncCoverageTab.vue.
-- Source of truth: ArcadiaCredentialState (per-client/per-credential row), enriched with
-- any extra credentials only present in ArcadiaCredential. Status values come straight
-- from those rows; MFA detection uses the same keyword list DSS_API uses.
WITH client_filter AS (
    SELECT
        c.IdClient
    FROM dbo.Client c
    WHERE (
        '{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
        OR c.IdClient IN (
            SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
        )
    )
),
credential_rows AS (
    -- Prefer ArcadiaCredentialState (one row per credential); fall back to ArcadiaCredential
    -- for credentials that exist there but not in state yet.
    SELECT
        cs.IdClient,
        cs.ArcadiaCredentialId,
        cs.Status,
        cs.StatusDetail,
        ISNULL(cs.IsCustomerActionRequired, 0) AS IsCustomerActionRequired
    FROM arcadia.ArcadiaCredentialState cs
    INNER JOIN client_filter cf ON cf.IdClient = cs.IdClient
    WHERE cs.ArcadiaCredentialId IS NOT NULL

    UNION

    SELECT
        ac.IdClient,
        ac.ArcadiaCredentialId,
        ac.Status,
        ac.StatusDetail,
        ISNULL(ac.IsCustomerActionRequired, 0) AS IsCustomerActionRequired
    FROM arcadia.ArcadiaCredential ac
    INNER JOIN client_filter cf ON cf.IdClient = ac.IdClient
    WHERE ac.ArcadiaCredentialId IS NOT NULL
        AND NOT EXISTS (
            SELECT 1 FROM arcadia.ArcadiaCredentialState cs2
            WHERE cs2.IdClient = ac.IdClient
                AND cs2.ArcadiaCredentialId = ac.ArcadiaCredentialId
        )
),
deduped AS (
    -- One row per (IdClient, ArcadiaCredentialId), picking any non-null Status/StatusDetail
    SELECT
        IdClient,
        ArcadiaCredentialId,
        MAX(Status) AS Status,
        MAX(StatusDetail) AS StatusDetail,
        MAX(CAST(IsCustomerActionRequired AS INT)) AS IsCustomerActionRequired
    FROM credential_rows
    GROUP BY IdClient, ArcadiaCredentialId
)
SELECT
    SUM(CASE WHEN Status IN ('SUCCESS', 'CONNECTION_SUCCESS') THEN 1 ELSE 0 END) AS connected_in_arcadia,
    SUM(CASE WHEN Status = 'CONNECTION_IN_PROGRESS' THEN 1 ELSE 0 END) AS pending,
    SUM(CASE
        WHEN Status = 'CONNECTION_FAILURE'
         AND NOT (
            LOWER(ISNULL(StatusDetail, '')) LIKE '%mfa%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multi-factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multi_factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multifactor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%two-factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%two_factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%otp%'
         )
        THEN 1 ELSE 0
    END) AS connection_failures,
    SUM(CASE
        WHEN (
            IsCustomerActionRequired = 1
            OR Status = 'CONNECTION_FAILURE'
        ) AND (
            LOWER(ISNULL(StatusDetail, '')) LIKE '%mfa%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multi-factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multi_factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%multifactor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%two-factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%two_factor%'
            OR LOWER(ISNULL(StatusDetail, '')) LIKE '%otp%'
        )
        THEN 1 ELSE 0
    END) AS mfa_failures,
    COUNT(1) AS total
FROM deduped;
