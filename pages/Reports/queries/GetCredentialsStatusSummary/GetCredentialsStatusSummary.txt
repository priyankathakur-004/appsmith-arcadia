SELECT
    cs.Status AS status,
    COUNT(1) AS cnt
FROM arcadia.ArcadiaCredentialState cs
WHERE cs.Status IS NOT NULL
  AND (
    '{{CredClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CredClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
GROUP BY cs.Status
ORDER BY cs.Status;
