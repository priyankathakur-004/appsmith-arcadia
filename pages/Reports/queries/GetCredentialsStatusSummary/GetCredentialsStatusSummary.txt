SELECT
    ac.Status AS status,
    COUNT(1) AS cnt
FROM dbo.ArcadiaCredential ac
WHERE ac.Status IS NOT NULL
  AND (
    '{{CredClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR ac.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CredClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
GROUP BY ac.Status
ORDER BY ac.Status;
