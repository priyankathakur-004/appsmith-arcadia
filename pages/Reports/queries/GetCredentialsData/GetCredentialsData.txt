WITH state_credentials AS (
    SELECT
        cs.IdClient,
        c.ClientName,
        cs.ArcadiaCredentialId,
        cs.CorrelationId,
        cs.ProviderId,
        cs.Status,
        cs.StatusDetail,
        cs.IsActive,
        cs.IsCustomerActionRequired,
        cs.NextScheduledAccountRunAt,
        cs.LastCredentialSyncedOn,
        cs.LastAccountSyncedOn,
        cs.LastStatusSource
    FROM arcadia.ArcadiaCredentialState cs
    INNER JOIN dbo.Client c ON c.IdClient = cs.IdClient
    WHERE 1=1
      AND (
        '{{CredClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
        OR cs.IdClient IN (
          SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CredClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
        )
      )
),
credential_groups AS (
    SELECT
        ac.IdClient,
        CASE
            WHEN ac.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(ac.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(ac.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(ac.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(ac.ArcadiaCredentialId)))
        END AS ArcadiaCredentialId,
        COUNT(DISTINCT ac.IdClientAccount) AS LocalAccountCount,
        MAX(ac.SyncedOn) AS LastLocalCredentialSyncedOn
    FROM arcadia.ArcadiaCredential ac
    WHERE ac.ArcadiaCredentialId IS NOT NULL
      AND LTRIM(RTRIM(ac.ArcadiaCredentialId)) <> ''
    GROUP BY ac.IdClient,
        CASE
            WHEN ac.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(ac.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(ac.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(ac.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(ac.ArcadiaCredentialId)))
        END
),
account_groups AS (
    SELECT
        aca.IdClient,
        CASE
            WHEN aca.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(aca.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(aca.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(aca.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(aca.ArcadiaCredentialId)))
        END AS ArcadiaCredentialId,
        COUNT(DISTINCT aca.ArcadiaAccountId) AS ArcadiaAccountCount,
        COUNT(DISTINCT CASE WHEN ISNULL(aca.IsActive, 0) = 1 THEN aca.ArcadiaAccountId END) AS ActiveArcadiaAccountCount,
        COUNT(DISTINCT CASE WHEN aca.IdClientAccount IS NOT NULL THEN aca.ArcadiaAccountId END) AS HasClientAccountCount,
        MAX(aca.LastSyncedOn) AS LastAccountSyncedOn
    FROM arcadia.ArcadiaCredentialAccount aca
    WHERE aca.ArcadiaCredentialId IS NOT NULL
      AND LTRIM(RTRIM(aca.ArcadiaCredentialId)) <> ''
    GROUP BY aca.IdClient,
        CASE
            WHEN aca.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(aca.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(aca.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(aca.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(aca.ArcadiaCredentialId)))
        END
)
SELECT
    sc.ArcadiaCredentialId AS id,
    sc.ArcadiaCredentialId AS credentialId,
    sc.CorrelationId AS correlationId,
    sc.Status AS status,
    sc.Status AS credentialStatus,
    sc.StatusDetail AS statusDetail,
    sc.StatusDetail AS credentialStatusDetail,
    sc.ProviderId AS providerId,
    sc.IdClient AS idClient,
    sc.ClientName AS client_name,
    sc.IsActive AS isActive,
    sc.IsCustomerActionRequired AS isCustomerActionRequired,
    sc.NextScheduledAccountRunAt AS nextScheduledAccountRunAt,
    sc.NextScheduledAccountRunAt AS nextRunAt,
    ISNULL(cg.LocalAccountCount, 0) AS localAccountCount,
    ISNULL(ag.ArcadiaAccountCount, 0) AS arcadiaAccountCount,
    ISNULL(ag.ArcadiaAccountCount, 0) AS totalAccounts,
    ISNULL(ag.ActiveArcadiaAccountCount, 0) AS activeArcadiaAccountCount,
    ISNULL(ag.HasClientAccountCount, 0) AS hasClientAccountCount,
    ISNULL(ag.ArcadiaAccountCount, 0) - ISNULL(ag.HasClientAccountCount, 0) AS missingClientAccountCount,
    COALESCE(sc.LastCredentialSyncedOn, cg.LastLocalCredentialSyncedOn) AS lastCredentialSyncedOn,
    COALESCE(sc.LastCredentialSyncedOn, cg.LastLocalCredentialSyncedOn) AS syncedOn,
    COALESCE(sc.LastAccountSyncedOn, ag.LastAccountSyncedOn) AS lastAccountSyncedOn,
    sc.LastStatusSource AS lastStatusSource
FROM state_credentials sc
LEFT JOIN credential_groups cg
    ON cg.IdClient = sc.IdClient
    AND cg.ArcadiaCredentialId = sc.ArcadiaCredentialId
LEFT JOIN account_groups ag
    ON ag.IdClient = sc.IdClient
    AND ag.ArcadiaCredentialId = sc.ArcadiaCredentialId
WHERE (
    '{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR sc.Status IN (
      SELECT value FROM STRING_SPLIT('{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
ORDER BY sc.ClientName, sc.ArcadiaCredentialId
OFFSET {{(CredentialsTable.pageNo - 1) * 50}} ROWS
FETCH NEXT 50 ROWS ONLY;
