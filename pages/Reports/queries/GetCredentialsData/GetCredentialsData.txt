SELECT
    ac.ArcadiaCredentialId AS arcadia_credential_id,
   
    ac.AccountNumber AS account_number,
    ac.Status AS status,
    ac.StatusDetail AS status_detail,
    ac.ProviderId AS provider_id,
    ac.IdClient AS id_client,
    c.ClientName AS client_name,
    ac.SyncedOn AS synced_on
FROM dbo.ArcadiaCredential ac
INNER JOIN dbo.Client c ON ac.IdClient = c.IdClient
WHERE 1=1
  AND (
    '{{CredClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR ac.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CredClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
  AND (
    '{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR ac.Status IN (
      SELECT value FROM STRING_SPLIT('{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
  
ORDER BY c.ClientName, ac.AccountNumber
OFFSET {{(CredentialsTable.pageNo - 1) * 50}} ROWS
FETCH NEXT 50 ROWS ONLY;
