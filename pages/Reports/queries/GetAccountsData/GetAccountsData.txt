WITH ranked_accounts AS (
    SELECT
        aca.IdClient,
        c.ClientName,
        aca.IdClientAccount,
        aca.AccountNumber,
        aca.NormalizedAccountNumber,
        aca.ArcadiaAccountId,
        aca.ArcadiaCredentialId,
        aca.ProviderId,
        aca.ProviderName,
        aca.LatestStatementDate,
        aca.NextExpectedPostDate,
        aca.Status,
        aca.StatusDetail,
        aca.AccountType,
        ISNULL(aca.IsExcluded, 0) AS IsExcluded,
        aca.IsActive,
        aca.MissingSince,
        aca.LastSeenOn,
        aca.LastSyncedOn,
        ROW_NUMBER() OVER (
            PARTITION BY
                CASE
                    WHEN aca.IdClientAccount IS NOT NULL THEN CONCAT('client:', CONVERT(NVARCHAR(50), aca.IdClientAccount))
                    WHEN aca.ArcadiaAccountId IS NOT NULL THEN CONCAT('arcadia:', aca.ArcadiaAccountId)
                    ELSE CONCAT(
                        'fallback:',
                        ISNULL(CONVERT(NVARCHAR(50), aca.IdClient), ''),
                        ':',
                        ISNULL(aca.ArcadiaCredentialId, ''),
                        ':',
                        ISNULL(aca.AccountNumber, ''),
                        ':',
                        ISNULL(aca.ProviderId, '')
                    )
                END
            ORDER BY
                CASE WHEN ISNULL(aca.IsActive, 0) = 1 THEN 0 ELSE 1 END,
                ISNULL(aca.LastSyncedOn, aca.LastSeenOn) DESC,
                aca.LastSeenOn DESC,
                aca.LatestStatementDate DESC,
                aca.ArcadiaCredentialId DESC,
                aca.ArcadiaAccountId DESC
        ) AS RowNum
    FROM arcadia.ArcadiaCredentialAccount aca
    INNER JOIN dbo.Client c ON c.IdClient = aca.IdClient
    WHERE 1=1
      AND (
        '{{AccClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
        OR aca.IdClient IN (
          SELECT CAST(value AS INT) FROM STRING_SPLIT('{{AccClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
        )
      )
      AND (
        '{{AccStatusSelect.selectedOptionValues?.join(",") || ""}}' = ''
        OR aca.Status IN (
          SELECT value FROM STRING_SPLIT('{{AccStatusSelect.selectedOptionValues?.join(",") || ""}}', ',')
        )
      )
      AND (
        '{{AccOnlyMatchedCheckbox?.isChecked ? "1" : "0"}}' = '0'
        OR aca.IdClientAccount IS NOT NULL
      )
      AND (
        '{{AccHasVendorCredCheckbox?.isChecked ? "1" : "0"}}' = '0'
        OR EXISTS (
          SELECT 1 FROM dbo.ClientAccount ca
          WHERE ca.IdClientAccount = aca.IdClientAccount
            AND ca.IdVendorCredential IS NOT NULL
        )
      )
)
SELECT
    ra.IdClient AS idClient,
    ra.ClientName AS client_name,
    ra.IdClientAccount AS idClientAccount,
    ra.AccountNumber AS accountNumber,
    ra.NormalizedAccountNumber AS normalizedAccountNumber,
    ra.ArcadiaAccountId AS id,
    ra.ArcadiaAccountId AS accountId,
    ra.ArcadiaCredentialId AS credentialId,
    ra.ProviderId AS providerId,
    ra.ProviderName AS providerName,
    ra.LatestStatementDate AS latestStatementDate,
    ra.NextExpectedPostDate AS nextExpectedPostDate,
    ra.Status AS status,
    ra.Status AS accountStatus,
    ra.StatusDetail AS statusDetail,
    ra.StatusDetail AS accountStatusDetail,
    ra.AccountType AS type,
    ra.AccountType AS accountType,
    ra.IsExcluded AS isExcluded,
    ra.IsActive AS isActive,
    ra.MissingSince AS missingSince,
    ra.LastSeenOn AS lastSeenOn,
    ra.LastSyncedOn AS lastSyncedOn,
    ra.LastSyncedOn AS syncedOn
FROM ranked_accounts ra
WHERE ra.RowNum = 1
ORDER BY ra.ClientName, ra.AccountNumber, ra.ArcadiaCredentialId, ra.ArcadiaAccountId
OFFSET {{(AccountsTable.pageNo - 1) * 50}} ROWS
FETCH NEXT 50 ROWS ONLY;
