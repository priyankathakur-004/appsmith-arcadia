WITH UniqueAccounts AS (
    SELECT 
        c.ProviderClientId,
        ca.AccountNumber_Raw,

        ROW_NUMBER() OVER (
            PARTITION BY c.ProviderClientId, ca.AccountNumber_Raw
            ORDER BY ca.IdClientAccount
        ) AS rn

    FROM dbo.ClientAccount ca
    INNER JOIN dbo.Client c 
        ON ca.IdClient = c.IdClient
    INNER JOIN dbo.Vendor v 
        ON v.IdVendor = ca.IdVendor 
        AND v.IsInactive = 0
    INNER JOIN dbo.VendorCredential vc 
        ON vc.IdVendorCredential = ca.IdVendorCredential

    WHERE ca.IsInactive = 0
      AND c.IdProvider = 7

      AND (
        {{ this.params.clientId ? 1 : 0 }} = 0
        OR c.ProviderClientId = '{{ this.params.clientId }}'
      )
)

SELECT COUNT(*) AS total_count
FROM UniqueAccounts
WHERE rn = 1;
