-- Accounts summary mirroring DSS_API _build_simple_account_items bucket-priority logic.
-- For each active DSS account, classify into a bucket using the same priority chain DSS_API uses,
-- then aggregate. Numbers should be close to (not always pixel-identical with) DSS_UI; a few
-- DSS_API branches (intent rows, customer_action_required reasons beyond MFA, manual access paths)
-- are approximated.
WITH client_filter AS (
    SELECT c.IdClient
    FROM dbo.Client c
    WHERE c.IdProvider = 7
      AND (
        '{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
        OR c.IdClient IN (
          SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
        )
      )
),
accounts AS (
    SELECT
        ca.IdClient,
        ca.IdClientAccount,
        ca.IdVendor,
        ca.IdVendorCredential,
        ca.AccountNumber_Raw,
        -- Normalize: prefer DS-side AccountNumber, fall back to raw; strip spaces/dashes/punct, upper-case.
        UPPER(LTRIM(RTRIM(
            REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(REPLACE(
                REPLACE(REPLACE(REPLACE(REPLACE(
                    COALESCE(NULLIF(LTRIM(RTRIM(ca.AccountNumber)), ''), ca.AccountNumber_Raw, '')
                , ' ', ''), '-', ''), '_', ''), '.', '')
            , '/', ''), '\', ''), ':', ''), ','  , '')
            , '(', ''), ')', ''), '+', ''), '#', '')
        ))) AS NormalizedAccountNumber
    FROM dbo.ClientAccount ca
    INNER JOIN client_filter cf ON cf.IdClient = ca.IdClient
    INNER JOIN dbo.Vendor v ON v.IdVendor = ca.IdVendor AND v.IsInactive = 0
    WHERE ca.IsInactive = 0
),
plans AS (
    SELECT
        ap.IdClientAccount,
        ap.IdClient,
        ap.ReasonCode,
        ap.AccessPathType,
        ap.PlanStatus,
        NULLIF(LTRIM(RTRIM(ap.ArcadiaProviderId)), '') AS PlanArcadiaProviderId
    FROM arcadia.ArcadiaAccountPlan ap
    INNER JOIN client_filter cf ON cf.IdClient = ap.IdClient
),
provider_by_vendor AS (
    SELECT
        pv.IdVendor,
        MAX(NULLIF(LTRIM(RTRIM(pv.ArcadiaProviderId)), '')) AS ArcadiaProviderId
    FROM arcadia.ArcadiaProviders pv
    GROUP BY pv.IdVendor
),
direct_active AS (
    -- accounts with at least one ACTIVE direct ArcadiaCredentialAccount row
    SELECT aca.IdClient, aca.IdClientAccount, COUNT(1) AS Cnt
    FROM arcadia.ArcadiaCredentialAccount aca
    INNER JOIN client_filter cf ON cf.IdClient = aca.IdClient
    WHERE aca.IdClientAccount IS NOT NULL
      AND ISNULL(aca.IsActive, 0) = 1
    GROUP BY aca.IdClient, aca.IdClientAccount
),
direct_any AS (
    -- accounts with any direct ArcadiaCredentialAccount row (active or historical)
    SELECT aca.IdClient, aca.IdClientAccount, COUNT(1) AS Cnt
    FROM arcadia.ArcadiaCredentialAccount aca
    INNER JOIN client_filter cf ON cf.IdClient = aca.IdClient
    WHERE aca.IdClientAccount IS NOT NULL
    GROUP BY aca.IdClient, aca.IdClientAccount
),
same_number_unmatched AS (
    -- DSS_API only triggers this if the unmatched row belongs to a credential the account
    -- already knows about. Approximated here: there must be a credential linking us AND
    -- the unmatched account number row must also belong to that same credential.
    SELECT DISTINCT a.IdClient, a.IdClientAccount
    FROM accounts a
    INNER JOIN arcadia.ArcadiaCredentialAccount mine
        ON mine.IdClient = a.IdClient
        AND mine.IdClientAccount = a.IdClientAccount
    INNER JOIN arcadia.ArcadiaCredentialAccount sibling
        ON sibling.IdClient = mine.IdClient
        AND sibling.ArcadiaCredentialId = mine.ArcadiaCredentialId
        AND ISNULL(sibling.IsActive, 0) = 1
        AND sibling.NormalizedAccountNumber = a.NormalizedAccountNumber
        AND (sibling.IdClientAccount IS NULL OR sibling.IdClientAccount <> a.IdClientAccount)
    WHERE a.NormalizedAccountNumber IS NOT NULL
      AND NOT EXISTS (
          SELECT 1 FROM direct_active da
          WHERE da.IdClient = a.IdClient AND da.IdClientAccount = a.IdClientAccount
      )
),
matched_credential_state AS (
    -- DSS_API: when an account has an active direct match, check the MATCHED credential's
    -- state (not other unrelated credentials). Prefer active matches; fall back to historical.
    SELECT
        a.IdClient,
        a.IdClientAccount,
        MAX(CASE WHEN cs.Status = 'CONNECTION_FAILURE' THEN 1 ELSE 0 END) AS HasConnectionFailure,
        MAX(CASE WHEN cs.Status = 'CONNECTION_IN_PROGRESS' THEN 1 ELSE 0 END) AS HasConnectionInProgress,
        MAX(CASE WHEN ISNULL(cs.IsCustomerActionRequired, 0) = 1 THEN 1 ELSE 0 END) AS HasCustomerActionRequired
    FROM accounts a
    INNER JOIN arcadia.ArcadiaCredentialAccount aca
        ON aca.IdClient = a.IdClient
        AND aca.IdClientAccount = a.IdClientAccount
        AND ISNULL(aca.IsActive, 0) = 1
    INNER JOIN arcadia.ArcadiaCredentialState cs
        ON cs.IdClient = aca.IdClient
        AND cs.ArcadiaCredentialId = aca.ArcadiaCredentialId
    GROUP BY a.IdClient, a.IdClientAccount
),
historical_credential_state AS (
    -- Same but including historical (non-active) matches — used when direct_active is empty
    SELECT
        a.IdClient,
        a.IdClientAccount,
        MAX(CASE WHEN cs.Status = 'CONNECTION_FAILURE' THEN 1 ELSE 0 END) AS HasConnectionFailure,
        MAX(CASE WHEN cs.Status = 'CONNECTION_IN_PROGRESS' THEN 1 ELSE 0 END) AS HasConnectionInProgress,
        MAX(CASE WHEN ISNULL(cs.IsCustomerActionRequired, 0) = 1 THEN 1 ELSE 0 END) AS HasCustomerActionRequired
    FROM accounts a
    INNER JOIN arcadia.ArcadiaCredentialAccount aca
        ON aca.IdClient = a.IdClient
        AND aca.IdClientAccount = a.IdClientAccount
    INNER JOIN arcadia.ArcadiaCredentialState cs
        ON cs.IdClient = aca.IdClient
        AND cs.ArcadiaCredentialId = aca.ArcadiaCredentialId
    GROUP BY a.IdClient, a.IdClientAccount
),
has_create_log AS (
    SELECT DISTINCT
        a.IdClient,
        a.IdClientAccount
    FROM arcadia.ArcadiaCredentialCreateLog l
    INNER JOIN accounts a
        ON a.IdClient = l.IdClientDS
        AND a.IdVendor = l.IdVendorDS
        AND a.NormalizedAccountNumber = l.AccountNumber
),
classified AS (
    SELECT
        a.IdClient,
        a.IdClientAccount,
        a.NormalizedAccountNumber,
        a.IdVendor,
        COALESCE(pl.PlanArcadiaProviderId, pv.ArcadiaProviderId) AS ProviderId,
        CASE
            -- 1. Direct active arcadia match → check MATCHED credential's state
            WHEN ISNULL(da.Cnt, 0) > 0 AND ISNULL(mcs.HasConnectionFailure, 0) = 1 THEN 'failed'
            WHEN ISNULL(da.Cnt, 0) > 0 AND ISNULL(mcs.HasCustomerActionRequired, 0) = 1 THEN 'failed'
            WHEN ISNULL(da.Cnt, 0) > 0 AND ISNULL(mcs.HasConnectionInProgress, 0) = 1 THEN 'in_progress'
            WHEN ISNULL(da.Cnt, 0) > 0 THEN 'found'
            -- 2. Same-number-elsewhere (unmatched) → PENDING (found_not_matched)
            WHEN snu.IdClientAccount IS NOT NULL THEN 'pending'
            -- 3. Has direct historical match but no active → FAILED (previously seen now missing)
            WHEN ISNULL(dany.Cnt, 0) > 0 THEN 'failed'
            -- 4. No direct match, but related-credential signals
            WHEN ISNULL(hcs.HasConnectionFailure, 0) = 1 THEN 'failed'
            WHEN ISNULL(hcs.HasCustomerActionRequired, 0) = 1 THEN 'failed'
            WHEN ISNULL(hcs.HasConnectionInProgress, 0) = 1 THEN 'in_progress'
            -- 5. Plan-level reasons
            WHEN pl.ReasonCode = 'provider_not_mapped' THEN 'provider_not_mapped'
            WHEN pl.ReasonCode = 'vendor_credential_missing' THEN 'failed'
            WHEN pl.ReasonCode IN (
                'provider_metadata_missing','provider_required_fields_missing',
                'provider_required_fields_empty','unsupported_required_fields',
                'credential_required_fields_missing','statements_not_supported',
                'resolved_credentials_missing'
            ) THEN 'failed'
            WHEN pl.AccessPathType IN ('provider_specific','manual','unsupported') THEN 'failed'
            -- 6. No provider mapping at all
            WHEN COALESCE(pl.PlanArcadiaProviderId, pv.ArcadiaProviderId) IS NULL
              OR LTRIM(RTRIM(COALESCE(pl.PlanArcadiaProviderId, pv.ArcadiaProviderId))) = ''
              THEN 'provider_not_mapped'
            -- 7. Has create log but never linked → FAILED (credential_create_failed)
            WHEN hcl.IdClientAccount IS NOT NULL THEN 'failed'
            -- 8. waiting_on_account_sync: vendor credential set + provider mapped + no failures
            --    above → in_progress (account is awaiting Arcadia sync)
            WHEN a.IdVendorCredential IS NOT NULL
              AND COALESCE(pl.PlanArcadiaProviderId, pv.ArcadiaProviderId) IS NOT NULL
              THEN 'in_progress'
            -- 9. Default → FAILED (credential_not_created)
            ELSE 'failed'
        END AS Bucket
    FROM accounts a
    LEFT JOIN plans pl ON pl.IdClient = a.IdClient AND pl.IdClientAccount = a.IdClientAccount
    LEFT JOIN provider_by_vendor pv ON pv.IdVendor = a.IdVendor
    LEFT JOIN direct_active da ON da.IdClient = a.IdClient AND da.IdClientAccount = a.IdClientAccount
    LEFT JOIN direct_any dany ON dany.IdClient = a.IdClient AND dany.IdClientAccount = a.IdClientAccount
    LEFT JOIN same_number_unmatched snu ON snu.IdClient = a.IdClient AND snu.IdClientAccount = a.IdClientAccount
    LEFT JOIN matched_credential_state mcs ON mcs.IdClient = a.IdClient AND mcs.IdClientAccount = a.IdClientAccount
    LEFT JOIN historical_credential_state hcs ON hcs.IdClient = a.IdClient AND hcs.IdClientAccount = a.IdClientAccount
    LEFT JOIN has_create_log hcl ON hcl.IdClient = a.IdClient AND hcl.IdClientAccount = a.IdClientAccount
),
duplicate_groups AS (
    -- DSS_UI groups by (IdClient, IdVendor, NormalizedAccountNumber). Count of GROUPS with >1 member.
    SELECT COUNT(1) AS GroupCount
    FROM (
        SELECT IdClient, IdVendor, NormalizedAccountNumber
        FROM classified
        WHERE NormalizedAccountNumber IS NOT NULL
          AND LTRIM(RTRIM(NormalizedAccountNumber)) <> ''
        GROUP BY IdClient, IdVendor, NormalizedAccountNumber
        HAVING COUNT(1) > 1
    ) g
)
SELECT
    COUNT(1) AS total_accounts,
    (SELECT GroupCount FROM duplicate_groups) AS grouped_accounts,
    SUM(CASE WHEN Bucket = 'provider_not_mapped' THEN 0 ELSE 1 END) AS eligible_accounts,
    SUM(CASE WHEN Bucket = 'found' THEN 1 ELSE 0 END) AS linked_in_arcadia,
    SUM(CASE WHEN Bucket = 'in_progress' THEN 1 ELSE 0 END) AS in_progress,
    SUM(CASE WHEN Bucket = 'pending' THEN 1 ELSE 0 END) AS pending,
    SUM(CASE WHEN Bucket = 'failed' THEN 1 ELSE 0 END) AS failed,
    SUM(CASE WHEN Bucket = 'provider_not_mapped' THEN 1 ELSE 0 END) AS provider_not_mapped
FROM classified;
