SELECT COUNT(1) AS total
FROM arcadia.ArcadiaCredentialState cs
WHERE 1=1
  AND (
    '{{CredClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CredClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
  AND (
    '{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.Status IN (
      SELECT value FROM STRING_SPLIT('{{CredStatusSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
  )
  AND (
    '{{CredOnlyWithCorrelationCheckbox?.isChecked ? "1" : "0"}}' = '0'
    OR cs.CorrelationId IS NOT NULL
  )
  AND (
    '{{CredHideAlreadyExistingCheckbox?.isChecked ? "1" : "0"}}' = '0'
    OR NOT EXISTS (
      SELECT 1 FROM arcadia.ArcadiaCredentialCreateLog cl
      WHERE cl.IdClient = cs.IdClient
        AND cl.ArcadiaCredentialId = cs.ArcadiaCredentialId
        AND cl.Status = 'ALREADY_EXISTS'
    )
  );
