WITH credential_accounts AS (
    SELECT
        aca.IdClient,
        CASE
            WHEN aca.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(aca.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(aca.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(aca.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(aca.ArcadiaCredentialId)))
        END AS ArcadiaCredentialId,
        COUNT(DISTINCT aca.ArcadiaAccountId) AS ArcadiaAccountCount,
        COUNT(DISTINCT CASE WHEN ISNULL(aca.IsActive, 0) = 1 THEN aca.ArcadiaAccountId END) AS ActiveArcadiaAccountCount,
        COUNT(DISTINCT CASE WHEN aca.IdClientAccount IS NOT NULL THEN aca.ArcadiaAccountId END) AS HasClientAccountCount,
        MAX(aca.LastSyncedOn) AS LastAccountSyncedOn
    FROM arcadia.ArcadiaCredentialAccount aca
    WHERE aca.ArcadiaCredentialId IS NOT NULL
      AND LTRIM(RTRIM(aca.ArcadiaCredentialId)) <> ''
    GROUP BY aca.IdClient,
        CASE
            WHEN aca.ArcadiaCredentialId IS NULL OR LTRIM(RTRIM(aca.ArcadiaCredentialId)) = '' THEN NULL
            WHEN LEFT(LTRIM(RTRIM(aca.ArcadiaCredentialId)), 4) = 'crd_' THEN LTRIM(RTRIM(aca.ArcadiaCredentialId))
            ELSE CONCAT('crd_', LTRIM(RTRIM(aca.ArcadiaCredentialId)))
        END
)
SELECT
    cs.ArcadiaCredentialId AS credentialId,
    cs.IdClient AS idClient,
    c.ClientName AS clientName,
    cs.ProviderId AS providerId,
    cs.CorrelationId AS correlationId,
    cs.Status AS status,
    cs.StatusDetail AS statusDetail,
    cs.IsCustomerActionRequired AS isCustomerActionRequired,
    cs.IsActive AS isActive,
    ISNULL(ca.ArcadiaAccountCount, 0) AS arcadiaAccountCount,
    ISNULL(ca.ActiveArcadiaAccountCount, 0) AS activeArcadiaAccountCount,
    ISNULL(ca.HasClientAccountCount, 0) AS hasClientAccountCount,
    ISNULL(ca.ArcadiaAccountCount, 0) - ISNULL(ca.HasClientAccountCount, 0) AS missingClientAccountCount,
    cs.LastCredentialSyncedOn AS lastCredentialSyncedOn,
    cs.LastAccountSyncedOn AS lastAccountSyncedOn,
    cs.NextScheduledAccountRunAt AS nextScheduledAccountRunAt,
    CASE
        WHEN ISNULL(ca.ActiveArcadiaAccountCount, 0) > 0 THEN 'Connected'
        WHEN cs.Status = 'CONNECTION_IN_PROGRESS' THEN 'Pending'
        WHEN cs.Status IN ('CONNECTION_FAILURE', 'CONNECTION_DEACTIVATED') THEN 'Failed'
        ELSE 'Pending'
    END AS bucket
FROM arcadia.ArcadiaCredentialState cs
INNER JOIN dbo.Client c ON c.IdClient = cs.IdClient
LEFT JOIN credential_accounts ca
    ON ca.IdClient = cs.IdClient
    AND ca.ArcadiaCredentialId = cs.ArcadiaCredentialId
WHERE (
    '{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}' = ''
    OR cs.IdClient IN (
      SELECT CAST(value AS INT) FROM STRING_SPLIT('{{CoverageClientSelect.selectedOptionValues?.join(",") || ""}}', ',')
    )
)
ORDER BY c.ClientName, cs.ArcadiaCredentialId
OFFSET {{(CoverageCredentialsTable.pageNo - 1) * 50}} ROWS
FETCH NEXT 50 ROWS ONLY;
